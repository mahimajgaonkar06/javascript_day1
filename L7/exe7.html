} // Kilometers to Miles else if (from === "km" && to === "miles") { return `${length / 1.6} miles`; } // Miles to Feet else if (from === "miles" && to === "ft") { return `${length * 5280} ft`; } // Kilometers to Feet else if (from === "km" && to === "ft") { return `${length * 3281} ft`; } // Feet to Miles else if (from === "ft" && to === "miles") { return `${length / 5280} miles`; } // Feet to Kilometers else if (from === "ft" && to === "km") { return `${length / 3281} km`; } // Same unit else if (from === to) { return `${length} ${to}`; } // Invalid unit else { return `Invalid unit: ${from}`; } } // 7g examples console.log(convertLength(50, "miles", "km")); console.log(convertLength(32, "km", "miles")); console.log(convertLength(50, "km", "km"));
